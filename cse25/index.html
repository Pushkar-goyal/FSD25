const form = document.getElementById("requestForm"); async function loadRequests() { const response = await fetch("/api/requests"); const requests = await response.json(); const container = document.getElementById("requests"); container.innerHTML = ""; if (requests.length === 0) { container.innerHTML = '

No requests submitted yet.

'; return; } requests.forEach(request => { container.innerHTML += `

${request.category}

Name: ${request.name}

Email: ${request.email}

Problem: ${request.description}

Priority: ${request.priority}

`; }); } form.addEventListener("submit", async (event) => { event.preventDefault(); const id = document.getElementById("requestId").value; const data = { name: document.getElementById("name").value, email: document.getElementById("email").value, category: document.getElementById("category").value, description: document.getElementById("description").value, priority: document.getElementById("priority").value }; if (id) { await fetch(`/api/requests/${id}`, { method: "PUT", headers: { "Content-Type": "application/json" }, body: JSON.stringify(data) }); } else { await fetch("/api/requests", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify(data) }); } resetForm(); loadRequests(); }); async function editRequest(id) { const response = await fetch(`/api/requests/${id}`); const request = await response.json(); document.getElementById("requestId").value = request.id; document.getElementById("name").value = request.name; document.getElementById("email").value = request.email; document.getElementById("category").value = request.category; document.getElementById("description").value = request.description; document.getElementById("priority").value = request.priority; window.scrollTo({ top: 0, behavior: "smooth" }); } async function deleteRequest(id) { await fetch(`/api/requests/${id}`, { method: "DELETE" }); loadRequests(); } function resetForm() { form.reset(); document.getElementById("requestId").value = ""; } loadRequests();
